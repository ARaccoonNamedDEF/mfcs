-- >> ESC [pseudo-class]; entity state checker
do
    MFCS.ESC = ESC
    local Instances = {}
    
    -- Instance
    function ESC.Create(Ent)
        if Instances[Ent] then return Instances[Ent] end
        
        local ent = {}
        
        local valid = Ent:isValid()
        local hookName = "mfcs_esc-" .. Ent:entIndex()
        
        local function OnEntityRemoved(InEnt)
            if InEnt != Ent then return end
            
            valid = false
            hook.run(hookName)
            hook.remove("EntityRemoved", hookName)
        end
        
        -- # Setup
        
        hook.add("EntityRemoved", hookName, OnEntityRemoved)
        
        -- # Accessors
        
        function ent:GetIsValid() return valid end
        
        function ent:GetHookName() return hookName end
        
        Instances[Ent] = ent
        return ent
    end
    
    -- Returns entity checker object
    function ESC:GetObject(Ent) return Instances[Ent] end
end

-- >> ELIST; connected entity list
do
    local Connections = {}
    local PurgeEntities = Empty
    
    -- Set initial entity list
    function ELIST:SetEntities(Tab)
        Connections = Tab
        //if CLIENT then PurgeEntities() end
    end
    
    -- Remove hooks from EntityRemoved
    // FUSED OFF
    function ELIST:RemoveHooks()
        if SERVER then return end
        
        for _, ent in ipairs(Connections) do
            hook.remove(ESC.Create(ent):GetHookName(), "elist_purge")
        end
    end
    
    -- Remove destroyed entities from list
    // FUSED OFF
    function PurgeEntities()
        local newConnections = {}
        for _, ent in ipairs(Connections) do
            local obj = ESC.Create(ent)
            if obj:GetIsValid() then
                hook.add(obj:GetHookName(), "elist_purge", PurgeEntities)
                table.insert(newConnections, ent)
            end
        end
        Connections = newConnections
    end
    
    -- # Accessors
    
    // This is very un-ideal; it removes invalid entities from Connections EVERY time it's called
    // Reason being (depending on starfall version), an invalid entity in a trace filter cause 'Object no longer valid'
    // Well, CLIENT doesn't seem to mind too much
    function MFCS:GetConnections()
        if SERVER then return Connections end
        
        local newConnections = {}
        for _, ent in ipairs(Connections) do
            if not ent:isValid() then continue end
            table.insert(newConnections, ent)
        end
        
        Connections = newConnections
        return Connections
    end
end

-- >> Cord [pseudo-class]; networked table
do
    Cord.OnReceive = Empty
    
    local Instances = {}; local Queue = {}
    local HookName = ""
    local HookImmediateName = ""
    
    local CordList = {}
    local ChannelList = {}
    
    local OnReceive = Empty; local OnReceiveImmediate = Empty
    
    -- Cord instance
    function Cord.CreateCord(ID)
        local cord = {}
        CordList[ID] = cord
        
        cord.OnReceive = Empty
        local dataList = {}
        local keyList = {}
        local writeOperationList = {}
        local readOperationList = {}
        
        local ClearVars = Empty
        local AddVar = Empty; local AutoAdd = Empty
        local AddWrite = Empty; local AddRead = Empty
        
        -- # Methods
        
        -- Set variables from table
        function cord:AddFromTable(Table)
            ClearVars()
            
            if table.isSequential(Table) then
                for _, var in ipairs(Table) do
                    AutoAdd(var)
                end
            else
                for k, var in pairs(Table) do
                    AutoAdd(var, k)
                end
            end
        end
        
        -- Queue data, sending done via channel
        function cord:Queue(Channel)
            Channel:Queue(ID)
        end
        
        -- Send instance Data immediately
        function cord:SendImmediate(Destination)
            net.start(HookImmediateName)
                net.writeString(ID)
                self:WriteAll()
            net.send(Destination)
        end
        
        -- Write outgoing data
        function cord:WriteAll()
            for _, writeOperation in ipairs(writeOperationList) do
                writeOperation()
            end
        end
        
        -- Read incoming data & call OnReceive
        function cord:ReadAll(Length, Sender)
            for _, readOperation in ipairs(readOperationList) do
                readOperation()
            end
            
            self.OnReceive(Length, Sender)
        end
        
        -- # Accessors
        
        function cord:GetVarIndex(Index) return dataList[Index] end
        function cord:SetVarIndex(Index, Var) dataList[Index] = Var end
        
        function cord:GetVarKey(Key)
            local index = keyList[Key]
            return dataList[index]
        end
        
        function cord:SetVarKey(Key, Var)
            local index = keyList[Key]
            dataList[index] = Var
        end
        
        function cord:GetData() return dataList end
        
        -- Operations on values here do not work, also returns a new table everytime
        function cord:GetKeyedData()
            local keyedDataList = {}
            for k, index in pairs(keyList) do
                keyedDataList[k] = dataList[index]
            end
            
            return keyedDataList
        end
        
        -- # Typed write, read
        
        function cord:AddAuto(Default, Name)
            return AutoAdd(Default, Name)
        end
        
        function cord:AddBool(Default, Name)
            local index = AddVar(Default or false, Name)
            AddWrite(function(Var) net.writeBool(Var) end)
            AddRead(function() return net.readBool() end)
            return index
        end
        
        function cord:AddInt(Default, Name)
            local index = AddVar(Default or 0, Name)
            AddWrite(function(Var) net.writeInt(Var, 32) end)
            AddRead(function() return net.readInt(32) end)
            return index
        end
        
        function cord:AddFloat(Default, Name)
            local index = AddVar(Default or 0.0, Name)
            AddWrite(function(Var) net.writeFloat(Var, 32) end)
            AddRead(function() return net.readFloat(32) end)
            return index
        end
        
        function cord:AddString(Default, Name)
            local index = AddVar(Default or "", Name)
            AddWrite(function(Var) net.writeString(Var) end)
            AddRead(function() return net.readString() end)
            return index
        end
        
        function cord:AddEntity(Default, Name)
            local index = AddVar(Default or NULL, Name)
            
            -- # This complains about entities being NULL, despite it being fine with manually being passed NULL???
            AddWrite(
                function(Var)
                    if Var:isValid() then
                        net.writeEntity(Var)
                        return
                    end
                    net.writeEntity(NULL)
                end
            )
            
            AddRead(function() return net.readEntity() end)
            return index
        end
        
        function cord:AddVector(Default, Name)
            local index = AddVar(Default or Vector(0, 0, 0), Name)
            AddWrite(function(Var) net.writeVector(Var) end)
            AddRead(function() return net.readVector() end)
            return index
        end
        
        function cord:AddAngle(Default, Name)
            local index = AddVar(Default or Angle(0, 0, 0), Name)
            AddWrite(function(Var) net.writeAngle(Var) end)
            AddRead(function() return net.readAngle() end)
            return index
        end
        
        -- Please, like, never ever used this
        function cord:AddTable(Default, Name)
            local index = AddVar(Default or {}, Name)
            AddWrite(function(Var) net.writeTable(Var) end)
            AddRead(function() return net.readTable() end)
            return index
        end
        
        -- # Helpers
        
        function AddVar(Var, Name)
            local varIndex = table.insert(dataList, Var)
            if Name != nil then keyList[Name] = varIndex end
            return varIndex
        end
        
        function AddWrite(Func)
            local index = #dataList
            table.insert(writeOperationList, function() Func(dataList[index]) end)
        end
        
        function AddRead(Func)
            local index = #dataList
            table.insert(readOperationList, function() dataList[index] = Func() end)
        end
        
        function ClearVars()
            table.empty(dataList)
            table.empty(keyList)
            table.empty(writeOperationList)
            table.empty(readOperationList)
        end
        
        function AutoAdd(Var, Name)
            local typeString = type(Var)
            
            if typeString == "boolean" then
                return cord:AddBool(Var, Name)
            end
            
            if typeString == "number" then
                return cord:AddFloat(Var, Name)
            end
            
            if typeString == "string" then
                return cord:AddString(Var, Name)
            end
            
            if typeString == "Entity" or typeString == "Player" then
                return cord:AddEntity(Var, Name)
            end
            
            if typeString == "Vector" then
                return cord:AddVector(Var, Name)
            end
            
            if typeString == "Angle" then
                return cord:AddAngle(Var, Name)
            end
            
            if typeString == "table" then
                return cord:AddTable(Var, Name)
            end
        end
        
        return cord
    end
    
    -- Channel instance
    function Cord.CreateChannel()
        local channel = {}
        table.insert(ChannelList, channel)
        
        local cordList = {}
        local cordLookup = {}
        
        -- Send cord data to server or client(s)
        function channel:SendAll(Destination)
            if #cordList == 0 then return end
            
            net.start(HookName)
                net.writeInt(#cordList, 32)
                
                for _, id in pairs(cordList) do
                    net.writeString(id)
                end
                
                for _, cordIndex in pairs(cordList) do
                    CordList[cordIndex]:WriteAll()
                end
            net.send(Destination)
            
            self:Dump()
        end
        
        -- Queue cord to be sent
        function channel:Queue(CordID)
            if cordLookup[CordID] then return end
            table.insert(cordList, CordID)
            cordLookup[CordID] = true
        end
        
        -- Empty cordList
        function channel:Dump()
            table.empty(cordList)
            table.empty(cordLookup)
        end
        
        return channel
    end
    
    function Cord:Initialize(Prefix)
        HookName = string.format("%s_cord", Prefix)
        HookImmediateName = string.format("%s-immediate", HookName)
        
        net.receive(HookName, OnReceive)
        net.receive(HookImmediateName, OnReceiveImmediate)
    end
    
    -- Update data on all queued objects
    function OnReceive(Length, Player)
        local count = net.readInt(32)
        
        local readList = {}
        for I = 1, count do
            table.insert(readList, CordList[net.readString()])
        end
        
        for _, cord in ipairs(readList) do
            cord:ReadAll(Length, Player)
        end
        
        Cord.OnReceive(Length, Player)
    end
    
    -- Update data on single object
    function OnReceiveImmediate()
        CordList[net.readString()]:ReadAll()
    end
end

-- >> EXIO; extra inputs
do
    local Defaults = {
        Boolean = false,
        Number = 0,
        String = "",
        Entity = NULL,
        Vector = Vector(),
        Angle = Angle(),
        Array = {},
        Table = {},
    }
    
    local InputTypeList; local OutputTypeList;
    local InputCordList = {}; local OutputCordList = {}
    local IOChannel = Cord.CreateChannel()
    local MarkerCord = Cord.CreateCord("mfcs-exio_marker")
    
    local PopulateDefault = Empty
    
    -- # Input, output creation
    
    function EXIO:AddInput(Name, Type)
        if InputCordList[Name] then return end
        
        local cord = Cord.CreateCord(string.format("mfcs-exio_in-%s", Name))
        PopulateDefault(cord, Name, Type)
        InputCordList[Name] = cord
        
        if SERVER then
            table.insert(InputTypeList, { Name = Name, Type = Type })
        end
    end
    
    function EXIO:AddOutput(Name, Type)
        if OutputCordList[Name] then return end
        
        local cord = Cord.CreateCord(string.format("mfcs-exio_out-%s", Name))
        PopulateDefault(cord, Name, Type)
        OutputCordList[Name] = cord
        
        if SERVER then
            table.insert(OutputTypeList, { Name = Name, Type = Type })
            
            cord.OnReceive = function(_, Sender)
                wire.ports[Name] = cord:GetVarIndex(1)
            end
        end
    end
    
    -- Specify type for Array & Table, auto for others
    function PopulateDefault(CordObj, Name, Type)
        if Type != "Array" and Type != "Table" then
            CordObj:AddAuto(Defaults[Type], Name)
        else
            CordObj:AddTable(Defaults[Type], Name)
        end
    end
    
    -- # Accessors
    
    function EXIO:GetInput(Name) return InputCordList[Name]:GetVarIndex(1) end
    
    function EXIO:GetOutput(Name) return OutputCordList[Name]:GetVarIndex(1) end
    function EXIO:SetOutput(Name, Value)
        OutputCordList[Name]:SetVarIndex(1, Value)
        if CLIENT then OutputCordList[Name]:Queue(IOChannel) end
    end
    
    if SERVER then
        InputTypeList = {}; OutputTypeList = {}
        
        local QueueInput = Empty
        
        function EXIO:Initialize()
            for name, cord in pairs(InputCordList) do
                cord:SetVarIndex(1, wire.ports[name])
            end
        end
        
        function EXIO:MakeIO()
            for _, dataType in ipairs(InputTypeList) do
                MFCS.AddInput(dataType.Name, dataType.Type)
            end
            
            for _, dataType in ipairs(OutputTypeList) do
                MFCS.AddOutput(dataType.Name, dataType.Type)
            end
        end
        
        function EXIO:OnInput(Name, Value)
            local cord = InputCordList[Name]
            if not cord then return end
            
            cord:SetVarIndex(1, Value)
            cord:Queue(IOChannel)
        end
        
        -- Send IO to target
        function EXIO:SendIO(Target)
            MarkerCord:Queue(IOChannel)
            IOChannel:SendAll(Target)
        end
        
        -- Send all IO to target
        function EXIO:ResendAllIO(Target)
            IOChannel:Dump()
            
            for name, cord in pairs(InputCordList) do
                cord:Queue(IOChannel)
            end
            
            for name, cord in pairs(OutputCordList) do
                cord:Queue(IOChannel)
            end
            
            IOChannel:SendAll(Target)
        end
    end
    
    if CLIENT then
        MarkerCord.OnReceive = function()
            hook.run(MFCS.EVENT.PING_SERVER)
            IOChannel:SendAll(Target)
        end
    end
end

if SERVER then
    Assembly = {}; Operator = {}

    -- >> Setup functions
    do
        function MFCS:Initialize()
            Manager:MakeIO()
            Manager:Initialize()
        end
    end
    
    -- >> Manager; it manages things
    do
        local OnEnterChair = Empty; local OnPingDriver = Empty
        
        function Manager:Initialize()
            Assembly:Initialize()
            Operator:Initialize()
            hook.run(MFCS.EVENT.INITIALIZE)
            
            hook.add("PlayerEnteredVehicle", "mfcs-driver_in", function(Ply, Chair) OnEnterChair(Ply, Chair, true) end)
            hook.add("PlayerLeaveVehicle", "mfcs-driver_out", function(Ply, Chair) OnEnterChair(Ply, Chair, false) end)
            hook.add("Input", "mfcs_mng-input", function(Name, Value) EXIO:OnInput(Name, Value) end)
            hook.add("DupeFinished", "mfcs-reset", function() restart() end)
            timer.create("mfcs-ping_driver", 1 / Settings.NetworkRate, 0, OnPingDriver)
        end
        
        function Manager:MakeIO()
            Operator:MakeIO()
            hook.run(MFCS.EVENT.MAKE_IO)
            EXIO:MakeIO()
            
            Assembly:CreateIO()
        end
        
        -- On enter & exit chair
        function OnEnterChair(Player, Chair, Board)
            if not Operator:CheckChair(Chair) then return end
            
            if Board then
                Operator:SetDriver(Player)
                EXIO:ResendAllIO(Player)
            else
                Operator:SetDriver(NULL)
            end
            
            DriverChannel:Dump()
            
            DriverCord:SetVarKey("Active", Board)
            DriverCord:SetVarKey("Connections", MFCS.GetConnections())
            DriverCord:Queue(DriverChannel)
            
            DriverChannel:SendAll(Player)
            
            hook.run(MFCS.EVENT.BOARD_VEHICLE, Player, Board)
        end
        
        -- # Passthroughs
        
        function OnPingDriver()
            if Operator:GetDriver() == NULL then return end
            
            hook.run(MFCS.EVENT.PING_DRIVER)
            EXIO:SendIO(Operator:GetDriver())
        end
    end
    
    -- >> Assembly; io & misc functions for vehicle
    do
        local Inputs = {}; local InputTypes = {}; local InputLookup = {}
        local Outputs = {}; local OutputTypes = {}; local OutputLookup = {}
        
        local RunFilter = Empty
        
        function Assembly:Initialize()
            ELIST:SetEntities( RunFilter(chip():getAllConstrained()) )
        end
        
        -- Execute final io creation
        function Assembly:CreateIO()
            wire.adjustInputs(Inputs, InputTypes)
            wire.adjustOutputs(Outputs, OutputTypes)
        end
        
        -- Adds an input
        function MFCS.AddInput(Name, Type)
            if InputLookup[Name] then return false end
            
            InputLookup[Name] = true
            table.insert(Inputs, Name)
            table.insert(InputTypes, Type)
            return true
        end
        
        -- Adds an output
        function MFCS.AddOutput(Name, Type)
            if OutputLookup[Name] then return false end
            
            OutputLookup[Name] = true
            table.insert(Outputs, Name)
            table.insert(OutputTypes, Type)
            return true
        end
        
        -- Filter out entities from constraints
        function RunFilter(ConnectionList)
            local filter = {
                gmod_wire_hologram = true,
                starfall_hologram = true,
            }
            
            local filteredList = {}
            for _, connection in ipairs(ConnectionList) do
                if filter[connection:getClass()] then continue end
                table.insert(filteredList, connection)
            end
            
            return filteredList
        end
    end
    
    -- >> Operator [SV]; container for driver & chair
    do
        local Driver = NULL; local ChairEnt = NULL
        
        function Operator:Initialize()
            ChairEnt = wire.ports.Chair
        end
        
        function Operator:MakeIO()

            return ChairEnt:isValid() and ChairEnt == Chair
        end
        
        -- # Accessors
        
        function Operator:GetDriver() return Driver end
        function Operator:SetDriver(Value) Driver = Value end
        
        function MFCS.GetDriver() return Driver end
    end

    -- >> Requestor; permissions stuff
    do
        local Permissions = {}; local Lookup = {}
        local CanRequest = true; local Ready = false
        
        local RequestAnswer = Empty; local HasAllPerms = Empty
        
        -- Try to get the wanted permissions
        function Requestor:SendRequest()
            if not CanRequest then return end
            CanRequest = false
            
            if HasAllPerms() then
                Ready = true
                hook.run(EVENT.REQUEST_ANSWER, Ready)
            else
                setupPermissionRequest(Permissions, "Stuff needed for MFCS", true)
                hook.add("permissionrequest", "mfcs-req_answer", RequestAnswer)
                sendPermissionRequest()
            end
        end
        
        -- Adds permission to request
        function MFCS.AddPermission(Perm)
            if Lookup[Perm] then return end
            
            table.insert(Permissions, Perm)
            Lookup[Perm] = true
        end
        
        -- Process answer
        function RequestAnswer()
            Ready = permissionRequestSatisfied()
            hook.run(EVENT.REQUEST_ANSWER, Ready)
            hook.remove("permissionrequest", "mfcs-req_answer")
        end
        
        -- Check if user has perms already
        function HasAllPerms()
            for _, perm in ipairs(Permissions) do
                local has, reason = hasPermission(perm)
                if not has and reason != "Entity is invalid" then
                    return false
                end
            end
            
            return true
        end
        
        -- # Accessors
        
        function MFCS.GetReady() return Ready end
    end
    
    -- >> Keys; keeps track of & updates user input
    do
        local ScrollActive = false
        local Scroll = Empty; local Input = Empty
        
        -- Turns input listening on or off
        function Keys:SetLogging(Active)
            if Active then
                hook.add("inputPressed", "mfcs_core-down", function(Key) Input(Key, true) end)
                hook.add("inputReleased", "mfcs_core-up", function(Key) Input(Key, false) end)
                hook.add("mouseWheeled", "mfcs_core-scroll", Scroll)
                hook.add("tick", "mfcs_core-reset", function() ScrollActive = true end)
            else
                hook.remove("inputPressed", "mfcs_core-down")
                hook.remove("inputReleased", "mfcs_core-up")
                hook.remove("mouseWheeled", "mfcs_core-scroll")
                hook.remove("tick", "mfcs_core-reset")
            end
        end
        
        -- Ran in mouseWheeled; scrolling
        function Scroll(Dir)
            if not ScrollActive then return end
            
            ScrollActive = false
            hook.run(MFCS.EVENT.OPERATOR_SCROLL, Dir)
        end
        
        -- Run key functions
        function Input(Key, Down)
            hook.run(MFCS.EVENT.OPERATOR_KEY, Key, Down)
        end
    end
end
